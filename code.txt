using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Lab6_KASD
{
    public class Request
{
    public int RequestId { get; set; }
    public int Priority { get; set; }
    public int StepAdded { get; set; }
    public int StepRemoved { get; set; }
    
    public int WaitingTime 
    { 
        get { return StepRemoved - StepAdded; } 
    }
}

public class RequestComparer : IComparer<Request>
{
    public int Compare(Request x, Request y)
    {
        if (x == null && y == null) return 0;
        if (x == null) return 1;
        if (y == null) return -1;
        
        int priorityCompare = y.Priority.CompareTo(x.Priority);
        if (priorityCompare != 0)
            return priorityCompare;
            
        return x.StepAdded.CompareTo(y.StepAdded);
    }
}

public class MyPriorityQueue<T>
{
    private T[] queue;
    private int size;
    private IComparer<T> comparator;

    public MyPriorityQueue(IComparer<T> comp)
    {
        queue = new T[11];
        size = 0;
        comparator = comp;
    }

    public MyPriorityQueue()
    {
        queue = new T[11];
        size = 0;
        comparator = Comparer<T>.Default;
    }

    public void Add(T e)
    {
        if (e == null) return;

        if (size >= queue.Length)
        {
            int newCapacity = CalculateNewCapacity();
            ResizeQueue(newCapacity);
        }

        queue[size] = e;
        SiftUp(size);
        size++;
    }

    public void Clear()
    {
        Array.Clear(queue, 0, size);
        size = 0;
    }

    public bool IsEmpty()
    {
        return size == 0;
    }

    public int Size()
    {
        return size;
    }

    public T Poll()
    {
        if (size == 0)
            return default(T);

        T result = queue[0];
        RemoveAt(0);
        return result;
    }

    private void BuildHeap()
    {
        for (int i = size / 2 - 1; i >= 0; i--)
        {
            SiftDown(i);
        }
    }

    private void SiftUp(int index)
    {
        while (index > 0)
        {
            int parentIndex = (index - 1) / 2;
            if (comparator.Compare(queue[index], queue[parentIndex]) < 0)
            {
                Swap(index, parentIndex);
                index = parentIndex;
            }
            else
            {
                break;
            }
        }
    }

    private void SiftDown(int index)
    {
        while (index < size)
        {
            int leftChildIndex = 2 * index + 1;
            int rightChildIndex = 2 * index + 2;
            int swapIndex = index;

            if (leftChildIndex < size && comparator.Compare(queue[leftChildIndex], queue[swapIndex]) < 0)
                swapIndex = leftChildIndex;

            if (rightChildIndex < size && comparator.Compare(queue[rightChildIndex], queue[swapIndex]) < 0)
                swapIndex = rightChildIndex;

            if (swapIndex != index)
            {
                Swap(index, swapIndex);
                index = swapIndex;
            }
            else
            {
                break;
            }
        }
    }

    private void Swap(int index1, int index2)
    {
        T temp = queue[index1];
        queue[index1] = queue[index2];
        queue[index2] = temp;
    }

    private int CalculateNewCapacity()
    {
        if (queue.Length < 64)
            return queue.Length + 2;
        else
            return queue.Length + (queue.Length / 2);
    }

    private void ResizeQueue(int newCapacity)
    {
        T[] newQueue = new T[newCapacity];
        Array.Copy(queue, 0, newQueue, 0, size);
        queue = newQueue;
    }

    private void RemoveAt(int index)
    {
        if (index < 0 || index >= size) return;

        queue[index] = queue[size - 1];
        queue[size - 1] = default(T);
        size--;
        SiftDown(index);
    }
}

public class PriorityRequestSystem
{
    private MyPriorityQueue<Request> requestQueue;
    private int currentRequestId;
    private int currentStep;
    private StreamWriter logWriter;
    
    public PriorityRequestSystem()
    {
        requestQueue = new MyPriorityQueue<Request>(new RequestComparer());
        currentRequestId = 1;
        currentStep = 0;
        logWriter = new StreamWriter("log.txt"); // свой путь добавить не забудь
    }
    
    public void ProcessRequests(int N)
    {
        Console.WriteLine($"Начинаем обработку {N} шагов добавления заявок...");
        
        for (int step = 1; step <= N; step++)
        {
            currentStep = step;
            ProcessStep(true);
        }
        
        Console.WriteLine($"Завершена фаза добавления. Осталось заявок: {requestQueue.Size()}");
        
        int removalStep = N + 1;
        while (!requestQueue.IsEmpty())
        {
            currentStep = removalStep++;
            ProcessStep(false);
        }
        
        Console.WriteLine("Все заявки обработаны.");
    }
    
    private void ProcessStep(bool addRequests)
    {
        Random random = new Random();
        
        if (addRequests)
        {
            int requestsToAdd = random.Next(1, 11);
            
            for (int i = 0; i < requestsToAdd; i++)
            {
                Request newRequest = new Request
                {
                    RequestId = currentRequestId++,
                    Priority = random.Next(1, 6),
                    StepAdded = currentStep
                };
                
                requestQueue.Add(newRequest);
                LogAction("ADD", newRequest);
            }
            
            Console.WriteLine($"Шаг {currentStep}: Добавлено {requestsToAdd} заявок. Всего в очереди: {requestQueue.Size()}");
        }
        
        if (!requestQueue.IsEmpty())
        {
            Request removedRequest = requestQueue.Poll();
            removedRequest.StepRemoved = currentStep;
            LogAction("REMOVE", removedRequest);
            
            Console.WriteLine($"Шаг {currentStep}: Удалена заявка {removedRequest.RequestId} " +
                            $"(приоритет {removedRequest.Priority}, время ожидания {removedRequest.WaitingTime})");
        }
        else
        {
            Console.WriteLine($"Шаг {currentStep}: Очередь пуста, удаление невозможно");
        }
    }
    
    private void LogAction(string action, Request request)
    {
        string logEntry = $"{action} {request.RequestId} {request.Priority} {currentStep}";
        logWriter.WriteLine(logEntry);
        logWriter.Flush();
    }
    
    public void AnalyzeResults()
    {
        if (!File.Exists("log.txt"))
        {
            Console.WriteLine("Файл log.txt не найден");
            return;
        }
        
        Dictionary<int, Request> requests = new Dictionary<int, Request>();
        int maxWaitingTime = 0;
        Request longestWaitingRequest = null;
        
        string[] lines = File.ReadAllLines("log.txt");
        foreach (string line in lines)
        {
            string[] parts = line.Split(' ');
            if (parts.Length < 4) continue;
            
            string action = parts[0];
            int requestId = int.Parse(parts[1]);
            int priority = int.Parse(parts[2]);
            int step = int.Parse(parts[3]);
            
            if (action == "ADD")
            {
                if (!requests.ContainsKey(requestId))
                {
                    requests[requestId] = new Request 
                    { 
                        RequestId = requestId, 
                        Priority = priority, 
                        StepAdded = step 
                    };
                }
            }
            else if (action == "REMOVE")
            {
                if (requests.ContainsKey(requestId))
                {
                    requests[requestId].StepRemoved = step;
                    int waitingTime = requests[requestId].WaitingTime;
                    
                    if (waitingTime > maxWaitingTime)
                    {
                        maxWaitingTime = waitingTime;
                        longestWaitingRequest = requests[requestId];
                    }
                }
            }
        }
        
        Console.WriteLine("\nАНАЛИЗ РЕЗУЛЬТАТОВ");
        Console.WriteLine($"Общее количество заявок: {requests.Count}");
        Console.WriteLine($"Максимальное время ожидания: {maxWaitingTime} шагов");
        
        if (longestWaitingRequest != null)
        {
            Console.WriteLine("\nИнформация о заявке с максимальным временем ожидания:");
            Console.WriteLine($"Номер заявки: {longestWaitingRequest.RequestId}");
            Console.WriteLine($"Приоритет: {longestWaitingRequest.Priority}");
            Console.WriteLine($"Шаг добавления: {longestWaitingRequest.StepAdded}");
            Console.WriteLine($"Шаг удаления: {longestWaitingRequest.StepRemoved}");
            Console.WriteLine($"Время ожидания: {longestWaitingRequest.WaitingTime} шагов");
        }
    }
    
    public void Close()
    {
        logWriter?.Close();
    }
}

public class Program
{
    public static void Main()
    {
        Console.WriteLine("=== СИСТЕМА ОБРАБОТКИ ЗАЯВОК С ПРИОРИТЕТОМ ===");
        
        try
        {
            Console.Write("Введите количество шагов добавления заявок (N): ");
            int N = int.Parse(Console.ReadLine());
            
            if (N <= 0)
            {
                Console.WriteLine("Количество шагов должно быть положительным числом");
                return;
            }
            
            PriorityRequestSystem system = new PriorityRequestSystem();
            
            try
            {
                system.ProcessRequests(N);
                system.AnalyzeResults();
            }
            finally
            {
                system.Close();
            }
            
            Console.WriteLine("\nЛоги сохранены в файл log.txt");
        }
        catch (FormatException)
        {
            Console.WriteLine("Ошибка: Введите корректное число");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Произошла ошибка: {ex.Message}");
        }
        
        Console.WriteLine("Нажмите любую клавишу для выхода...");
        Console.ReadKey();
    }
}
