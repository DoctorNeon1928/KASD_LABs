    public interface IStudentOperations
    {
        void DisplayInfo();
        bool CheckAcademicStatus();
    }

    public interface ISubjectOperations
    {
        void DisplaySubjectInfo();
        void AddDepartment(string department);
    }

    public interface IInstituteOperations
    {
        void AddStudent(Student student);
        void RemoveStudent(Student student);
        List<Student> GetExcellentStudents();
    }

    public delegate void StudentActionDelegate(Student student);
    public delegate void SubjectActionDelegate(Subject subject);
    public delegate void InstituteActionDelegate(Institute institute);
    public class Student : IStudentOperations
    {
        public string LastName { get; set; }
        public string FirstName { get; set; }
        public int Course { get; set; }
        public string Group { get; set; }
        public List<int> Grades { get; set; }

        public Student(string lastName, string firstName, int course, string group, List<int> grades)
        {
            LastName = lastName;
            FirstName = firstName;
            Course = course;
            Group = group;
            Grades = grades;
        }


        public void DisplayInfo()
        {
            Console.WriteLine($"Студент: {LastName} {FirstName}, {Course} курс, группа {Group}");
            Console.Write("Оценки: ");
            for (int j = 0; j < Grades.Count; j++)
            {
                Console.Write(Grades[j]);
                if (j < Grades.Count - 1)
                    Console.Write(", ");
            }
            Console.WriteLine();
        }

        public bool CheckAcademicStatus()
        {
            return IsExcellent;
        }

        public bool IsExcellent
        {
            get
            {
                foreach (int grade in Grades)
                {
                    if (grade != 5)
                        return false;
                }
                return true;
            }
        }
    }

    public class Subject : ISubjectOperations
    {
        public string Name { get; set; }
        public List<string> Departments { get; set; } = new List<string>();

        public Subject(string name)
        {
            Name = name;
        }

        public void DisplaySubjectInfo()
        {
            Console.Write($"Предмет: {Name} (Кафедры: ");
            for (int i = 0; i < Departments.Count; i++)
            {
                Console.Write(Departments[i]);
                if (i < Departments.Count - 1)
                    Console.Write(", ");
            }
            Console.WriteLine(")");
        }

        public void AddDepartment(string department)
        {
            Departments.Add(department);
            Console.WriteLine($"Кафедра '{department}' добавлена к предмету '{Name}'");
        }
    }

    public class Institute : IInstituteOperations
    {
        public string Name { get; set; }
        public List<Student> Students { get; set; } = new List<Student>();
        public List<Subject> Subjects { get; set; } = new List<Subject>();

        public Institute(string name)
        {
            Name = name;
        }

        public void AddStudent(Student student)
        {
            Students.Add(student);
            Console.WriteLine($"Студент {student.LastName} {student.FirstName} добавлен в институт {Name}");
        }

        public void RemoveStudent(Student student)
        {
            if (Students.Remove(student))
            {
                Console.WriteLine($"Студент {student.LastName} {student.FirstName} удален из института {Name}");
            }
        }

        public List<Student> GetExcellentStudents()
        {
            List<Student> excellentStudents = new List<Student>();
            foreach (Student student in Students)
            {
                if (student.IsExcellent)
                {
                    excellentStudents.Add(student);
                }
            }
            return excellentStudents;
        }
    }

            int index = int.Parse(Console.ReadLine());

            if (index >= 0 && index < institutes.Count)
            {
                string deletedName = institutes[index].Name;
                institutes.RemoveAt(index);

                if (institutes.Count == 0)
                {
                    currentInstituteIndex = -1;
                }
                else if (currentInstituteIndex >= institutes.Count)
                {
                    currentInstituteIndex = institutes.Count - 1;
                }

                Console.WriteLine($"Институт '{deletedName}' удален!");
            }
            else
            {
                Console.WriteLine("Неверный номер института!");
            }
        }

        static void AddStudent(Institute institute)
        {

            string lastName = Console.ReadLine();

            string firstName = Console.ReadLine();
            Console.Write("Курс: ");
            int course = int.Parse(Console.ReadLine());
            Console.Write("Группа: ");
            string group = Console.ReadLine();

            Console.WriteLine("Введите оценки через пробел (2-5): ");
            string[] gradesInput = Console.ReadLine().Split(' ');
            List<int> grades = new List<int>();
            foreach (string gradeStr in gradesInput)
            {
                grades.Add(int.Parse(gradeStr));
            }

            Student student = new Student(lastName, firstName, course, group, grades);
            institute.Students.Add(student);
            Console.WriteLine("Студент добавлен!");
        }

        static void EditStudent(Institute institute)

                Console.Write("Имя: ");
                institute.Students[index].FirstName = Console.ReadLine();
                Console.Write("Курс: ");

                foreach (string gradeStr in gradesInput)

                    newGrades.Add(int.Parse(gradeStr));

                Console.WriteLine("Неверный индекс!");

        // Выполнение запроса варианта 5: Полный список отличников с указанием института, группы и курса
        static void ExecuteQuery()
        {
            if (institutes.Count == 0)
            {
                Console.WriteLine("Нет институтов для выполнения запроса!");
                return;
            }

            // Собираем всех отличников из всех институтов
            int totalExcellentStudents = 0;

            Console.WriteLine("=== Полный список отличников из всех институтов ===");
            foreach (Institute institute in institutes)
            {
                foreach (Student student in institute.Students)
                {
                    if (student.IsExcellent)
                    {
                        Console.WriteLine($"{student.LastName} {student.FirstName}, {institute.Name}, {student.Course} курс, гр. {student.Group}");
                        totalExcellentStudents++;
                    }
                }
            }

            if (totalExcellentStudents == 0)
            {
                Console.WriteLine("Отличников нет ни в одном институте!");
                return;
            }

        }
