using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Dursy_lab1_C__only
{
    class Program
    {
        static void Main()
        {
            
            string[] lines = File.ReadAllLines("input.txt");
            int n = int.Parse(lines[0]);

            
            double[] x = new double[n];
            string[] vectorValues = lines[1].Split(' ');
            for (int i = 0; i < n; i++)
            {
                x[i] = double.Parse(vectorValues[i]);
            }

            
            double[,] G = new double[n, n];
            for (int i = 0; i < n; i++)
            {
                string[] rowValues = lines[i + 2].Split(' ');
                for (int j = 0; j < n; j++)
                {
                    G[i, j] = double.Parse(rowValues[j]);
                }
            }

            
            for (int i = 0; i < n; i++)
            {
                for (int j = i + 1; j < n; j++)
                {
                    if (G[i, j] != G[j, i])
                    {
                        Console.WriteLine("Ошибка: матрица G не симметрична!");
                        return;
                    }
                }
            }

            // Вычисление длины: sqrt(x * G * x^T) ~ sqrt(sumI * sumJ xi * Gij * xj)
            double result = 0;
            for (int i = 0; i < n; i++)
            {
                double temp = 0;
                for (int j = 0; j < n; j++)
                {
                    temp += G[i, j] * x[j];
                }
                result += x[i] * temp;
            }

            double length = Math.Sqrt(result);
            Console.WriteLine($"Длина вектора: {length:F6}");
        }
          
    }
}
