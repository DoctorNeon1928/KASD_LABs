using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Lab2_KASD
{
    class ComplexNumber
    {
        public double Real;
        public double Imaginary;

        public ComplexNumber(double real = 0, double imaginary = 0)
        {
            Real = real;
            Imaginary = imaginary;
        }

        public void Print()
        {
            if (Imaginary >= 0)
                Console.WriteLine($"{Real} + {Imaginary}i");
            else
                Console.WriteLine($"{Real} - {-Imaginary}i");
        }

        public void Addition(double real, double imaginary)
        {
            Real += real;
            Imaginary += imaginary;
        }

        public void Subtraction(double real, double imaginary)
        {
            Real -= real;
            Imaginary -= imaginary;
        }

        public void Multiplication(double real, double imaginary)
        {
            double newReal = Real * real - Imaginary * imaginary;
            double newImaginary = Real * imaginary + Imaginary * real;
            Real = newReal;
            Imaginary = newImaginary;
        }

        public void Division(double real, double imaginary, double znamenatel)
        {
            double newReal = (Real * real + Imaginary * imaginary) / znamenatel;
            double newImaginary = (Imaginary * real - Real * imaginary) / znamenatel;
            Real = newReal;
            Imaginary = newImaginary;
        }

        public double Module()
        {
            return Math.Sqrt(Real * Real + Imaginary * Imaginary);
        }

        public double Argument()
        {
            return Math.Atan2(Imaginary, Real);
        }
    }

    class Program
    {
        static void Main()
        {
            ComplexNumber current = new ComplexNumber();
            bool running = true;

            Console.WriteLine("Калькулятор комплексных чисел");

            while (running)
            {
                Console.WriteLine("\nМеню:");
                Console.WriteLine("S - Установить число");
                Console.WriteLine("A - Сложение");
                Console.WriteLine("B - Вычитание");
                Console.WriteLine("M - Умножение");
                Console.WriteLine("D - Деление");
                Console.WriteLine("G - Модуль");
                Console.WriteLine("R - Аргумент");
                Console.WriteLine("P - Вывести число");
                Console.WriteLine("Q - Выход");
                Console.Write("Выберите команду: ");

                char command = Console.ReadKey().KeyChar;
                Console.WriteLine();

                switch (char.ToUpper(command))
                {
                    case 'S':
                        Console.Write("Введите вещественную часть: ");
                        double real = double.Parse(Console.ReadLine());
                        Console.Write("Введите мнимую часть: ");
                        double imaginary = double.Parse(Console.ReadLine());
                        current.Real = real;
                        current.Imaginary = imaginary;
                        Console.Write("Новое число: ");
                        current.Print();
                        break;

                    case 'A':
                        Console.Write("Введите вещественную часть второго числа: ");
                        real = double.Parse(Console.ReadLine());
                        Console.Write("Введите мнимую часть второго числа: ");
                        imaginary = double.Parse(Console.ReadLine());
                        current.Addition(real, imaginary);
                        Console.Write("Результат: ");
                        current.Print();
                        break;

                    case 'B':
                        Console.Write("Введите вещественную часть второго числа: ");
                        real = double.Parse(Console.ReadLine());
                        Console.Write("Введите мнимую часть второго числа: ");
                        imaginary = double.Parse(Console.ReadLine());
                        current.Subtraction(real, imaginary);
                        Console.Write("Результат: ");
                        current.Print();
                        break;

                    case 'M':
                        Console.Write("Введите вещественную часть второго числа: ");
                        real = double.Parse(Console.ReadLine());
                        Console.Write("Введите мнимую часть второго числа: ");
                        imaginary = double.Parse(Console.ReadLine());
                        current.Multiplication(real, imaginary);
                        Console.Write("Результат: ");
                        current.Print();
                        break;

                    case 'D':
                        Console.Write("Введите вещественную часть второго числа: ");
                        real = double.Parse(Console.ReadLine());
                        Console.Write("Введите мнимую часть второго числа: ");
                        imaginary = double.Parse(Console.ReadLine());
                        double znamenatel = real * real + imaginary * imaginary;
                        if (znamenatel == 0)
                        {
                            Console.WriteLine("Ошибка: деление на ноль!");
                        }
                        else
                        {
                            current.Division(real, imaginary, znamenatel);
                            Console.Write("Результат: ");
                            current.Print();
                        }
                        break;

                    case 'G':
                        Console.WriteLine($"Модуль: {current.Module()}");
                        break;

                    case 'R':
                        Console.WriteLine($"Аргумент: {current.Argument()} радиан");
                        break;

                    case 'P':
                        Console.Write("Текущее число: ");
                        current.Print();
                        break;

                    case 'Q':
                        running = false;
                        Console.WriteLine("Выход");
                        break;

                    default:
                        Console.WriteLine("Неизвестная команда");
                        break;
                }
            }
        }
    }
}
