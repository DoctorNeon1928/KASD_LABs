using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Lab2_Polymorphism
{
    public abstract class Organization
    {
        public string Name { get; set; }
        public string Address { get; set; }

        public Organization(string name = "Неизвестно", string address = "Неизвестно")
        {
            Name = name;
            Address = address;
        }

        public virtual void Display()
        {
            Console.WriteLine($"Организация: {Name}, Адрес: {Address}");
        }

        public abstract string GetName();
    }



    public interface IInsurable
    {
        void Insure(string objectName);
    }

    public interface IExtractable
    {
        void Resource();
    }

    public interface IProducible
    {
        void Product();
    }



    
    public class InsuranceCompany : Organization, IInsurable
    {
        public string Specialization { get; set; }

        public InsuranceCompany(string name, string address,
                                string specialization) : base(name, address)
        {
            Specialization = specialization;
        }

        public override void Display()
        {
            Console.WriteLine($"Страховая компания: {Name}");
            Console.WriteLine($"Адрес: {Address}");
            Console.WriteLine($"Специализация: {Specialization}");
        }

        public override string GetName()
        {
            return "Страховая компания";
        }

        public void Insure(string objectName)
        {
            Console.WriteLine($"{Name} застраховала: {objectName}");
        }
    }

    public class OilGasCompany : Organization, IExtractable
    {
        public int WellsCount { get; set; }

        public OilGasCompany(string name, string address,
                             int wellsCount) : base(name, address)
        {
            WellsCount = wellsCount;
        }

        public override void Display()
        {
            Console.WriteLine($"Нефтегазовая компания: {Name}");
            Console.WriteLine($"Адрес: {Address}");
            Console.WriteLine($"Количество скважин: {WellsCount}");
        }

        public override string GetName()
        {
            return "Нефтегазовая компания";
        }

        public void Resource()
        {
            Console.WriteLine($"{Name} добывает нефть и газ");
        }
    }

    public class Factory : Organization, IProducible
    {
        public string ProductType { get; set; }

        public Factory(string name, string address,
                        string productType) : base(name, address)
        {
            ProductType = productType;
        }

        public override void Display()
        {
            Console.WriteLine($"Завод: {Name}");
            Console.WriteLine($"Адрес: {Address}");
            Console.WriteLine($"Продукция: {ProductType}");
        }

        public override string GetName()
        {
            return "Завод";
        }

        public void Product()
        {
            Console.WriteLine($"{Name} производит: {ProductType}");
        }
    }

    class Program
    {
        static void Main(string[] args)
        {

            Organization org1 = new InsuranceCompany("Страх", "Москва", "Автострахование");
            Organization org2 = new OilGasCompany("РосНефть", "Чита", 250);
            Organization org3 = new Factory("АвтоВАЗ", "Тольятти", "Автомобили");


            List<Organization> orgs = new List<Organization> { org1, org2, org3 };

            //Полиморфизм
            foreach (Organization org in orgs)
            {
                org.Display();
                Console.WriteLine($"Тип организации: {org.GetName()}");
                Console.WriteLine();
            }

            Console.WriteLine("=== Демонстрация интерфейсов ===");

            if (org1 is IInsurable insurer)
            {
                insurer.Insure("здание бизнес-центра");
            }

            if (org2 is IExtractable extractor)
            {
                extractor.Resource();
            }

            if (org3 is IProducible producer)
            {
                producer.Product();
            }

            Console.ReadLine();
        }
    }
}
