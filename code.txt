using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Dursy_KASD_5
{
    public class BinaryHeap
    {
        private List<int> heap;
        private bool isMinHeap;

        public BinaryHeap(bool minHeap = true)
        {
            heap = new List<int>();
            isMinHeap = minHeap;
        }

        public BinaryHeap(int[] array, bool minHeap = true)
        {
            isMinHeap = minHeap;
            heap = new List<int>(array);
            BuildHeap();
        }

        public int Count
        {
            get { return heap.Count; }
        }

        public bool IsEmpty
        {
            get { return heap.Count == 0; }
        }

        public int Peek()
        {
            if (heap.Count == 0)
            {
                return 0;
            }

            return heap[0];
        }

        public void Add(int item)
        {
            heap.Add(item);
            SiftUp(heap.Count - 1);
        }

        public int Remove()
        {
            if (heap.Count == 0)
            {
                return 0;
            }

            int result = heap[0];
            int lastIndex = heap.Count - 1;
            heap[0] = heap[lastIndex];
            heap.RemoveAt(lastIndex);

            if (heap.Count > 0)
            {
                SiftDown(0);
            }

            return result;
        }

        public void UpdateKey(int index, int newValue)
        {
            if (index < 0 || index >= heap.Count)
            {
                return;
            }

            int oldValue = heap[index];
            heap[index] = newValue;

            if (newValue < oldValue)
            {
                SiftUp(index);
            }
            else if (newValue > oldValue)
            {
                SiftDown(index);
            }
        }

        public BinaryHeap Merge(BinaryHeap otherHeap)
        {
            if (otherHeap == null)
            {
                return new BinaryHeap(isMinHeap);
            }

            if (this.isMinHeap != otherHeap.isMinHeap)
            {
                return new BinaryHeap(isMinHeap);
            }

            int[] mergedArray = new int[this.heap.Count + otherHeap.heap.Count];
            this.heap.CopyTo(mergedArray, 0);
            otherHeap.heap.CopyTo(mergedArray, this.heap.Count);

            return new BinaryHeap(mergedArray, this.isMinHeap);
        }

        public int[] ToArray()
        {
            return heap.ToArray();
        }

        public void Clear()
        {
            heap.Clear();
        }

        private void BuildHeap()
        {
            for (int i = heap.Count / 2 - 1; i >= 0; i--)
            {
                SiftDown(i);
            }
        }

        private void SiftUp(int index)
        {
            while (index > 0)
            {
                int parentIndex = (index - 1) / 2;

                if (ShouldSwap(parentIndex, index))
                {
                    Swap(parentIndex, index);
                    index = parentIndex;
                }
                else
                {
                    break;
                }
            }
        }

        private void SiftDown(int index)
        {
            int lastIndex = heap.Count - 1;

            while (index < lastIndex)
            {
                int leftChildIndex = 2 * index + 1;
                int rightChildIndex = 2 * index + 2;
                int swapIndex = index;

                if (leftChildIndex <= lastIndex && ShouldSwap(swapIndex, leftChildIndex))
                {
                    swapIndex = leftChildIndex;
                }

                if (rightChildIndex <= lastIndex && ShouldSwap(swapIndex, rightChildIndex))
                {
                    swapIndex = rightChildIndex;
                }

                if (swapIndex != index)
                {
                    Swap(index, swapIndex);
                    index = swapIndex;
                }
                else
                {
                    break;
                }
            }
        }

        private bool ShouldSwap(int parentIndex, int childIndex)
        {
            if (isMinHeap)
            {
                return heap[parentIndex] > heap[childIndex];
            }
            else
            {
                return heap[parentIndex] < heap[childIndex];
            }
        }

        private void Swap(int index1, int index2)
        {
            int temp = heap[index1];
            heap[index1] = heap[index2];
            heap[index2] = temp;
        }
    }

    public class Program
    {
        public static void Main()
        {
            BinaryHeap minHeap = new BinaryHeap();

            minHeap.Add(5);
            minHeap.Add(3);
            minHeap.Add(8);
            minHeap.Add(1);
            minHeap.Add(10);

            Console.WriteLine("Min-Heap:");
            while (!minHeap.IsEmpty)
            {
                Console.WriteLine(minHeap.Remove());
            }

            BinaryHeap maxHeap = new BinaryHeap(false);

            maxHeap.Add(5);
            maxHeap.Add(3);
            maxHeap.Add(8);
            maxHeap.Add(1);
            maxHeap.Add(10);

            Console.WriteLine("\nMax-Heap:");
            while (!maxHeap.IsEmpty)
            {
                Console.WriteLine(maxHeap.Remove());
            }

            BinaryHeap heap1 = new BinaryHeap();
            heap1.Add(2);
            heap1.Add(4);
            heap1.Add(6);

            BinaryHeap heap2 = new BinaryHeap();
            heap2.Add(1);
            heap2.Add(3);
            heap2.Add(5);

            BinaryHeap mergedHeap = heap1.Merge(heap2);

            Console.WriteLine("\nMerged Heap:");
            while (!mergedHeap.IsEmpty)
            {
                Console.WriteLine(mergedHeap.Remove());
            }

            Console.WriteLine("\nПроверка UpdateKey:");
            BinaryHeap testHeap = new BinaryHeap();
            testHeap.Add(10);
            testHeap.Add(20);
            testHeap.Add(15);
            testHeap.Add(30);

            testHeap.UpdateKey(1, 5);
            Console.WriteLine("После обновления индексов 1-5:");
            while (!testHeap.IsEmpty)
            {
                Console.WriteLine(testHeap.Remove());
            }
        }
    }
}
