using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Dursy_Lab_8
{
    public class MyArrayList<T>
    {
        private T[] elementData;
        private int size;

        // 1) Конструктор для создания пустого динамического массива
        public MyArrayList()
        {
            elementData = new T[10];
            size = 0;
        }

        // 2) Конструктор для создания динамического массива из передаваемого массива
        public MyArrayList(T[] a)
        {
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                elementData = new T[10];
                size = 0;
                return;
            }

            elementData = new T[a.Length];
            Array.Copy(a, elementData, a.Length);
            size = a.Length;
        }

        // 3) Конструктор для создания пустого динамического массива с указанной емкостью
        public MyArrayList(int capacity)
        {
            if (capacity < 0)
            {
                Console.WriteLine("Ошибка: Емкость не может быть отрицательной. Установлена емкость по умолчанию: 10");
                capacity = 10;
            }

            elementData = new T[capacity];
            size = 0;
        }

        // 4) Метод для добавления элемента в конец динамического массива
        public void Add(T e)
        {
            EnsureCapacity(size + 1);
            elementData[size++] = e;
        }

        // 5) Метод для добавления элементов из массива
        public void AddAll(T[] a)
        {
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                return;
            }

            EnsureCapacity(size + a.Length);
            Array.Copy(a, 0, elementData, size, a.Length);
            size += a.Length;
        }

        // 6) Метод для удаления всех элементов из динамического массива
        public void Clear()
        {
            for (int i = 0; i < size; i++)
            {
                elementData[i] = default(T);
            }
            size = 0;
        }

        // 7) Метод для проверки наличия указанного объекта в динамическом массиве
        public bool Contains(object o)
        {
            if (o == null)
            {
                for (int i = 0; i < size; i++)
                {
                    if (elementData[i] == null)
                        return true;
                }
            }
            else
            {
                for (int i = 0; i < size; i++)
                {
                    if (o.Equals(elementData[i]))
                        return true;
                }
            }
            return false;
        }

        // 8) Метод для проверки наличия всех указанных объектов в динамическом массиве
        public bool ContainsAll(T[] a)
        {
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                return false;
            }

            foreach (var item in a)
            {
                if (!Contains(item))
                    return false;
            }
            return true;
        }

        // 9) Метод для проверки, является ли динамический массив пустым
        public bool IsEmpty()
        {
            return size == 0;
        }

        // 10) Метод для удаления указанного объекта из динамического массива
        public bool Remove(object o)
        {
            int index = IndexOf(o);
            if (index >= 0)
            {
                RemoveAt(index);
                return true;
            }
            return false;
        }

        // 11) Метод для удаления всех указанных объектов из динамического массива
        public void RemoveAll(T[] a)
        {
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                return;
            }

            foreach (var item in a)
            {
                while (Remove(item)) { }
            }
        }

        // 12) Метод для оставления в динамическом массиве только указанных объектов
        public void RetainAll(T[] a)
        {
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                return;
            }

            for (int i = size - 1; i >= 0; i--)
            {
                if (Array.IndexOf(a, elementData[i]) == -1)
                {
                    RemoveAt(i);
                }
            }
        }

        // 13) Метод для получения размера динамического массива в элементах
        public int Size()
        {
            return size;
        }

        // 14) Метод для возвращения массива объектов, содержащего все элементы динамического массива
        public object[] ToArray()
        {
            object[] result = new object[size];
            Array.Copy(elementData, result, size);
            return result;
        }

        // 15) Метод для возвращения массива объектов с возможностью использования переданного массива
        public T[] ToArray(T[] a)
        {
            if (a == null)
            {
                T[] result = new T[size];
                Array.Copy(elementData, result, size);
                return result;
            }

            if (a.Length < size)
            {
                T[] result = new T[size];
                Array.Copy(elementData, result, size);
                return result;
            }

            Array.Copy(elementData, a, size);
            if (a.Length > size)
            {
                a[size] = default(T);
            }
            return a;
        }

        // 16) Метод для добавления элемента в указанную позицию
        public void Add(int index, T e)
        {
            if (index < 0 || index > size)
            {
                Console.WriteLine($"Ошибка: Индекс {index} вне границ массива [0, {size}]");
                return;
            }

            EnsureCapacity(size + 1);
            if (index < size)
            {
                Array.Copy(elementData, index, elementData, index + 1, size - index);
            }
            elementData[index] = e;
            size++;
        }

        // 17) Метод для добавления элементов в указанную позицию
        public void AddAll(int index, T[] a)
        {
            if (index < 0 || index > size)
            {
                Console.WriteLine($"Ошибка: Индекс {index} вне границ массива [0, {size}]");
                return;
            }
            if (a == null)
            {
                Console.WriteLine("Ошибка: Переданный массив не может быть null");
                return;
            }

            int numNew = a.Length;
            EnsureCapacity(size + numNew);

            if (index < size)
            {
                Array.Copy(elementData, index, elementData, index + numNew, size - index);
            }

            Array.Copy(a, 0, elementData, index, numNew);
            size += numNew;
        }

        // 18) Метод для возвращения элемента в указанной позиции
        public T Get(int index)
        {
            if (index < 0 || index >= size)
            {
                Console.WriteLine($"Ошибка: Индекс {index} вне границ массива [0, {size - 1}]");
                return default(T);
            }

            return elementData[index];
        }

        // 19) Метод для возвращения индекса указанного объекта
        public int IndexOf(object o)
        {
            if (o == null)
            {
                for (int i = 0; i < size; i++)
                {
                    if (elementData[i] == null)
                        return i;
                }
            }
            else
            {
                for (int i = 0; i < size; i++)
                {
                    if (o.Equals(elementData[i]))
                        return i;
                }
            }
            return -1;
        }

        // 20) Метод для нахождения последнего вхождения указанного объекта
        public int LastIndexOf(object o)
        {
            if (o == null)
            {
                for (int i = size - 1; i >= 0; i--)
                {
                    if (elementData[i] == null)
                        return i;
                }
            }
            else
            {
                for (int i = size - 1; i >= 0; i--)
                {
                    if (o.Equals(elementData[i]))
                        return i;
                }
            }
            return -1;
        }

        // 21) Метод для удаления и возвращения элемента в указанной позиции
        public T RemoveAt(int index)
        {
            if (index < 0 || index >= size)
            {
                Console.WriteLine($"Ошибка: Индекс {index} вне границ массива [0, {size - 1}]");
                return default(T);
            }

            T oldValue = elementData[index];

            int numMoved = size - index - 1;
            if (numMoved > 0)
            {
                Array.Copy(elementData, index + 1, elementData, index, numMoved);
            }

            elementData[--size] = default(T);
            return oldValue;
        }

        // 22) Метод для замены элемента в указанной позиции новым элементом
        public T Set(int index, T e)
        {
            if (index < 0 || index >= size)
            {
                Console.WriteLine($"Ошибка: Индекс {index} вне границ массива [0, {size - 1}]");
                return default(T);
            }

            T oldValue = elementData[index];
            elementData[index] = e;
            return oldValue;
        }

        // 23) Метод для возвращения части динамического массива
        public MyArrayList<T> SubList(int fromIndex, int toIndex)
        {
            if (fromIndex < 0 || toIndex > size || fromIndex > toIndex)
            {
                Console.WriteLine($"Ошибка: Неверный диапазон [{fromIndex}, {toIndex}]");
                return new MyArrayList<T>();
            }

            int subSize = toIndex - fromIndex;
            T[] subArray = new T[subSize];
            Array.Copy(elementData, fromIndex, subArray, 0, subSize);
            return new MyArrayList<T>(subArray);
        }

        private void EnsureCapacity(int minCapacity)
        {
            if (minCapacity > elementData.Length)
            {
                int newCapacity = elementData.Length * 3 / 2 + 1;
                if (newCapacity < minCapacity)
                    newCapacity = minCapacity;

                T[] newElementData = new T[newCapacity];
                Array.Copy(elementData, newElementData, size);
                elementData = newElementData;
                Console.WriteLine($"Массив расширен: {elementData.Length} -> {newCapacity}");
            }
        }

        public void Print()
        {
            Console.Write("[");
            for (int i = 0; i < size; i++)
            {
                Console.Write(elementData[i]);
                if (i < size - 1)
                    Console.Write(", ");
            }
            Console.WriteLine($"] (размер: {size}, емкость: {elementData.Length})");
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("MyArrayList");

            Console.WriteLine("\n1. Создание пустого списка:");
            MyArrayList<int> list1 = new MyArrayList<int>();
            list1.Print();

            Console.WriteLine("\n2. Добавление элементов:");
            for (int i = 1; i <= 5; i++)
            {
                list1.Add(i * 10);
            }
            list1.Print();

            Console.WriteLine("\n3. Добавление элемента 25 на позицию 2:");
            list1.Add(2, 25);
            list1.Print();

            Console.WriteLine("\n4. Создание списка из массива:");
            string[] words = { "apple", "banana", "cherry" };
            MyArrayList<string> list2 = new MyArrayList<string>(words);
            list2.Print();

            Console.WriteLine("\n5. Проверка содержания элементов:");
            Console.WriteLine($"Содержит 'banana': {list2.Contains("banana")}");
            Console.WriteLine($"Содержит 'orange': {list2.Contains("orange")}");

            Console.WriteLine("\n6. Работа с элементами:");
            Console.WriteLine($"Элемент на позиции 1: {list2.Get(1)}");
            string oldValue = list2.Set(1, "blueberry");
            Console.WriteLine($"Замена 'banana' на 'blueberry'. Старое значение: {oldValue}");
            list2.Print();

            Console.WriteLine("\n7. Удаление элементов:");
            Console.WriteLine($"Удаление 'apple': {list2.Remove("apple")}");
            list2.Print();
            Console.WriteLine($"Удаление элемента на позиции 0: {list2.RemoveAt(0)}");
            list2.Print();

        }
    }
}
